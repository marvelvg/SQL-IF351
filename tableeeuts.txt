create table secretary(
     secid varchar(30) primary key,
     secname varchar(30),
     secoffice varchar(30)
);

create table professor(
	pid varchar(30),
	pname varchar(30),
	poffice varchar(30),
	psec varchar(30),

	key (psec),
	primary key(pid)
);

create table student(
	sid varchar(30) primary key,
	soffice varchar(30),
	scoffeemaker varchar(30),

	key(scoffeemaker)

);

create table coffeemaker(
	cmid varchar(30) primary key,
  	cmbrand varchar(30),
 	cmowner varchar(30),
	key(cmowner)
 );

create table workfor(
 	sid varchar(30),
 	pid varchar(30)
);

ALTER TABLE professor
ADD FOREIGN KEY (psec) REFERENCES secretary(secid);

ALTER TABLE student
 ADD FOREIGN KEY (scoffeemaker) REFERENCES coffeemaker(cmid);

ALTER TABLE coffeemaker
ADD FOREIGN KEY (cmowner) REFERENCES professor(pid);

ALTER TABLE workfor
    add FOREIGN KEY (sid) REFERENCES student(sid);

create table office(
 	studentshareoffice_id varchar(30),
 	professoroffice_id varchar(30),
 	secretaryoffice_id varchar(30)
);
