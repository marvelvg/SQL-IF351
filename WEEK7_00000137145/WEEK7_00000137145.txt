select 
	concat(e.nama_depan, ' ', coalesce(e.nama_belakang, '')) AS 'NAMA KARYAWAN', 
	group_concat(
		c.nama_depan, ' ', coalesce(c.nama_belakang, '') 
		ORDER BY c.nama_depan) AS 'NAMA CUSTOMER',
	sum(p.harga_paket) AS HARGA_PAKET

from employee e
JOIN transaction t 
	ON e.id_employee = t.id_employee
RIGHT JOIN customer c 
	ON t.id_customer = c.id_customer
RIGHT JOIN paket p 
	ON t.id_paket = p.id_paket


group by t.id_employee
order by harga_paket desc, concat(c.nama_depan, ' ', coalesce(c.nama_belakang, '')) asc
LIMIT 1
;
