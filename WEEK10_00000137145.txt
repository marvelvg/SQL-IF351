SOAL 1 - PROCEDURE DISCOUNT 

DELIMITER //

CREATE PROCEDURE diskon(IN percentage INT)
BEGIN
	SELECT 
		p.ProductName AS PRODUCT,
		CONCAT('Rp. ', ROUND(SUM(p.Price - p.Price * (percentage/100)))) AS HARGA
	FROM products p

	GROUP BY PRODUCT
	ORDER BY Price
	LIMIT 10;
END//
DELIMITER ;

MELO SUKA CEWE - MELOSUKACEWE

DELIMITER //

CREATE PROCEDURE hitung(IN kalimat VARCHAR(255))
BEGIN
    SELECT IF(TRIM(kalimat) = '', 0,
        LENGTH(TRIM(kalimat)) - LENGTH(REPLACE(TRIM(kalimat), ' ', '')) + 1
    ) AS TOTAL;
END //

DELIMITER ;

SOAL 3 - PRIMA CHECK
DELIMITER // 
	CREATE PROCEDURE cekPrima(IN number INT)

		BEGIN
		DECLARE i INT DEFAULT 2;
		DECLARE Togel BOOLEAN DEFAULT TRUE;

			IF number > 1 THEN
				WHILE i < number AND Togel = TRUE DO 
				  IF(number % i = 0) THEN SET Togel = FALSE;
				  END IF;

				  SET i = i + 1;
				END WHILE;				

				IF Togel = TRUE THEN 
					SELECT('IS PRIME NUMBER') AS HASIL;
				ELSE 
					SELECT('IS NOT PRIME NUMBER') AS HASIL;
				END IF;

			ELSE 
			 SELECT('MASUKIN ANGKA DI ATAS 1') AS 'Error Message';
			END IF;
		END //
DELIMITER ;

	DELIMITER //
		CREATE OR REPLACE PROCEDURE box(IN X INT, IN Y INT)
		  BEGIN
		    DECLARE horizontal_line VARCHAR(5000);
		    DECLARE vertical_line VARCHAR(5000);
		    DECLARE  total_rows INT;

		    SET horizontal_line = CONCAT('+', REPEAT(CONCAT(REPEAT('-', Y), '+'), X));
		    SET vertical_line = CONCAT('|', REPEAT(CONCAT(REPEAT(' ', Y), '|'), X));
	            SET total_rows = (X * (Y + 1)) + 1;

		    WITH RECURSIVE RowGenerator (n) AS (
			SELECT 1
			UNION ALL
			SELECT n + 1 FROM RowGenerator WHERE n < total_rows
		    )

		    SELECT
			CASE 
				WHEN (n - 1) % (Y + 1) = 0 THEN horizontal_line
				ELSE vertical_line
			END AS 'Box'
		    FROM RowGenerator;
		    

		  END //
		
	DELIMITER ;
