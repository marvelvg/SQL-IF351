CREATE PROCEDURE diskon(IN person INT)
BEGIN
	SELECT 
		p.ProductName AS PRODUCT
		
	FROM products p
	LEFT JOIN forderdetails f ON f.ArticleID = p.ArticleID
	GROUP BY PRODUCT;
