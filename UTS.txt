create table actors (
	id varchar(5) primary key,
	first_name varchar(15),
	last_name varchar(15),
	gender varchar(10),
	birth_date date

);

create table movies (
	id varchar(5) primary key,
	name varchar(40),
	year year,
	rank float	
);

create table roles (
	actor_id varchar(10) references actors(id),
	movie_id varchar(10) references movies(id),
	salary INT,
	
	primary key (actor_id, movie_id)
);

alter table roles
add column role varchar(40) AFTER movie_id;

from actors a

select concat(a.first_name, ' ', a.last_name) AS NAMA, sum(r.salary) AS SALARY
from actors a
join roles r on a.id = r.actor_id
group by NAMA
having SALARY > 4000000;

select concat(first_name, ' ', last_name) AS NAMA, birth_date AS DATE
from actors
where gender = 'M'
UNION 
select concat(first_name, ' ', last_name) AS NAMA, birth_date AS DATE
from actors 
group by NAMA
having YEAR(birth_date) < 1980;


create table calon(
	id_calon varchar(20) primary key,
	nama_calon varchar(30)
);

create table vote(
	id_calon varchar(20) references calon(id_calon)
,
	id_pemilih varchar(30) references pemilih(id_pemilih),

	primary key (id_calon, id_pemilih));

create table pemilih(
	id_pemilih varchar(20) primary key,
	nama_pemilih varchar(30),
	nomor_telepon_penduduk varchar(30)
);

from pemilih p
