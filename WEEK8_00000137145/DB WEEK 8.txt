select  concat(
RPAD(concat(e.FirstName, ' ', coalesce(e.LastName, '')), 40, ' '),

RPAD(
 CONCAT(
	(CASE WHEN LEFT(j.JobDesc, 1) IN ('A' , 'I' , 'U' , 'E' , 'O') THEN 'work as a '
	ELSE 'work as an '
	END)
	, 
j.JobDesc), 30, ' '),

LPAD(CONCAT(' In ', d.DeptName, ' Department'), 30, ' ')
) AS 'Employee Information'
from employees e

join jobs j on e.JobID = j.JobID
join depts d on d.DeptID = e.DeptID
;
